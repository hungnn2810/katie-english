BfaService methods (bfa.service.ts):
```typescript
analyze(audioBuffer, mimeType, word, expectedPhonemes) -> POST /analyze, timeout=120_000
  FormData fields: audio (filename audio.{ext}), word, expected_phonemes (JSON.stringify)

align(audioBuffer, mimeType, word, expectedPhonemes) -> POST /align, timeout=60_000
  FormData fields: audio, word, expected_phonemes (JSON.stringify)

transcribe(audioBuffer, mimeType) -> POST /transcribe, timeout=120_000
  FormData fields: audio only (no word, no expected_phonemes)

Extension logic (shared by all three):
  mimeType.includes('webm') -> 'webm'
  mimeType.includes('mp4')  -> 'mp4'
  else                      -> 'wav'
```

"""
Unit tests for BFA pure functions.
Run: cd bfa-service && python -m pytest test_bfa.py -v
No external services required — whisperx/fastapi/prometheus stubs applied via sys.modules.
"""
import sys
from types import ModuleType
from unittest.mock import MagicMock

# ── Stub all non-stdlib deps before importing main ──────────────────────────
# MagicMock handles attribute access, calls, and decorator usage transparently.
for _mod in [
    "whisperx",
    "bournemouth_aligner",
    "fastapi",
    "fastapi.responses",
    "prometheus_client",
]:
    if _mod not in sys.modules:
        sys.modules[_mod] = MagicMock()

import main  # noqa: E402 — must come after stubs

# ── _phoneme_cost ────────────────────────────────────────────────────────────

def test_phoneme_cost_identical():
    assert main._phoneme_cost("l", "l") == 0.0
    assert main._phoneme_cost("sh", "sh") == 0.0

def test_phoneme_cost_similar_pairs():
    """Each pair in _SIMILAR_PAIRS must cost exactly 0.5."""
    for pair in main._SIMILAR_PAIRS:
        a, b = tuple(pair)
        assert main._phoneme_cost(a, b) == 0.5, f"Expected 0.5 for ({a}, {b})"
        assert main._phoneme_cost(b, a) == 0.5, f"Expected 0.5 for ({b}, {a})"

def test_phoneme_cost_unrelated():
    assert main._phoneme_cost("l", "z") == 1.0
    assert main._phoneme_cost("a", "sh") == 1.0

# ── score_alignment ──────────────────────────────────────────────────────────

def test_score_alignment_perfect_match():
    score, ops = main.score_alignment(["c", "a", "t"], ["c", "a", "t"])
    assert score == 100
    assert len(ops) == 3
    assert all(op["status"] == "correct" for op in ops)

def test_score_alignment_similar_status_d01_regression():
    """D-01 fix: similar phoneme pair must produce status='similar', not 'substituted'."""
    score, ops = main.score_alignment(["r"], ["l"])
    assert len(ops) == 1
    assert ops[0]["status"] == "similar", (
        f"D-01 regression: expected 'similar' for r/l pair, got {ops[0]['status']!r}"
    )
    assert ops[0]["expected"] == "r"
    assert ops[0]["aligned"] == "l"
    # Score should be > 0 (similar is cheaper than full substitution)
    assert score > 0

def test_score_alignment_substituted_status():
    """Unrelated phoneme substitution must produce status='substituted'."""
    score, ops = main.score_alignment(["c"], ["z"])
    assert ops[0]["status"] == "substituted"
    assert score == 0

def test_score_alignment_missing_phonemes():
    """Phonemes in expected but not in aligned → 'missing' ops."""
    score, ops = main.score_alignment(["s", "h", "i", "p"], ["s", "h"])
    statuses = [op["status"] for op in ops]
    assert statuses.count("missing") == 2
    missing_expected = {op["expected"] for op in ops if op["status"] == "missing"}
    assert "i" in missing_expected
    assert "p" in missing_expected
    assert score < 100

def test_score_alignment_extra_phonemes():
    """Phonemes in aligned but not in expected → 'extra' ops."""
    score, ops = main.score_alignment(["c", "a", "t"], ["c", "a", "t", "s"])
    statuses = [op["status"] for op in ops]
    assert "extra" in statuses
    extra_ops = [op for op in ops if op["status"] == "extra"]
    assert extra_ops[0]["aligned"] == "s"
    assert extra_ops[0]["expected"] is None

def test_score_alignment_empty_expected():
    """Empty expected list with aligned phonemes → all extra ops, score=0."""
    score, ops = main.score_alignment([], ["a", "b"])
    assert score == 0
    assert all(op["status"] == "extra" for op in ops)

def test_score_alignment_empty_both():
    """Both lists empty → score=100, no ops."""
    score, ops = main.score_alignment([], [])
    assert score == 100
    assert ops == []

def test_score_alignment_mixed():
    """Mixed: correct + similar + missing in one word."""
    # "sh" → "sh" (correct), "i" → "e" (similar), "p" absent (missing)
    score, ops = main.score_alignment(["sh", "i", "p"], ["sh", "e"])
    statuses = [op["status"] for op in ops]
    assert "correct" in statuses
    assert "similar" in statuses
    assert "missing" in statuses
    assert 0 < score < 100

# ── normalize_ipa ────────────────────────────────────────────────────────────

def test_normalize_ipa_strips_stress():
    result = main.normalize_ipa("ˈæ")
    assert "ˈ" not in result

def test_normalize_ipa_strips_length():
    result = main.normalize_ipa("aː")
    assert "ː" not in result

def test_normalize_ipa_lowercases():
    result = main.normalize_ipa("A")
    assert result == result.lower()

# ── error_payload ────────────────────────────────────────────────────────────

def test_error_payload_shape():
    payload = main.error_payload("cat", "no speech detected")
    assert payload["success"] is False
    assert payload["word"] == "cat"
    assert payload["score"] == 0
    assert payload["phonemes"] == []
    assert len(payload["feedback"]) == 1
    assert payload["feedback"][0]["status"] == "error"
    assert "no speech detected" in payload["feedback"][0]["message"]
