      // Defined as a callback to avoid stale closure in finishSession
      const computeTotals = useCallback(() => {
        let total = 0, correct = 0;
        for (const a of activityStates) {
          if (a.type === 'MATCH') { total += a.pairs.length; correct += a.pairs.filter(p =&gt; p.status === 'locked').length; }

        setPageState('submitting');
        const { total, correct } = computeTotals();
        try {
          await saveReadingResult(sessionId, { correctItems: correct, totalItems: total });
          const completed = await completeSession(sessionId);
          setFinalResult(completed);
        } catch (err) {
          setSaveError(err instanceof Error ? err.message : 'Failed to save');
        }
        setPageState('results');
      }, [sessionId, computeTotals]);

      // ... load on mount

        if (sel == null) return;
        // word is identified by the pair.id of the pair that owns the word (i.e. pair.word belongs to pair.id);
        // a 'correct' match is selectedImageId === wordPairId.
        if (state.pairs.find(p =&gt; p.pair.id === wordPairId)?.status === 'locked') return;
        const isCorrect = sel === wordPairId;
        if (isCorrect) {
          onChange({
            ...state,
            selectedImageId: null,
            pairs: state.pairs.map(p =&gt; p.pair.id === wordPairId ? { ...p, status: 'locked' } : p),
          });
        } else {
          // Mark both shaking briefly, then return to idle (and clear selection)
          setState(prev =&gt; ({
            ...prev,

              pairs: prev.pairs.map(p =&gt; (p.pair.id === sel || p.pair.id === wordPairId) ? { ...p, status: 'idle' } : p),
