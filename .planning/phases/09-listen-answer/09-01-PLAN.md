    ```prisma
    model ListenItem {
      id           Int                @id @default(autoincrement())
      homeworkId   Int
      homework     Homework           @relation("ListenItems", fields: [homeworkId], references: [id], onDelete: Cascade)
      audioUrl     String
      keywords     String             // JSON array of expected keywords e.g. ["red", "cat"]
      expectedText String             // full expected answer for semantic scoring
      order        Int                @default(0)
      createdAt    DateTime           @default(now())
      results      ListenItemResult[]

      @@index([homeworkId])
      @@map("listen_items")
    }

    model ListenItemResult {
      id             Int             @id @default(autoincrement())
      sessionId      Int
      listenItemId   Int
      itemOrder      Int             @default(0)
      transcript     String          @default("")
      semanticScore  Float           @default(0)
      pronScore      Float           @default(0)
      compositeScore Float           @default(0)
      bfaFeedback    String?         // JSON — BFA feedback for matched keywords
      session        HomeworkSession @relation(fields: [sessionId], references: [id], onDelete: Cascade)
      listenItem     ListenItem      @relation(fields: [listenItemId], references: [id], onDelete: Cascade)

      @@map("listen_item_results")
    }
    ```

    Note on field names: CONTEXT.md D-03 uses `transcript` (not `transcribedText`), `pronScore` (not `pronunciationScore`), and `bfaFeedback` (optional String). Use these exact names from the CONTEXT spec since the game.service.ts plan will reference them.
